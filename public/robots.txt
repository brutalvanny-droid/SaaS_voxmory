# Règles d'exploration pour les moteurs de recherche.

User-agent: *
Allow: /

# Le studio d'administration est un espace privé : il n'a rien à faire
# dans un index de recherche. (Une règle « /#admin » ne servait à rien :
# tout ce qui suit un « # » n'est jamais envoyé au serveur.)
Disallow: /admin
Disallow: /api/

# Les espaces personnels sont déjà marqués « noindex » par la page
# elle-même ; on évite en plus de les explorer inutilement.
Disallow: /compte
Disallow: /mon-compte
